cmake_minimum_required(VERSION 3.16)
project(wave_linux_native_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
pkg_check_modules(NL REQUIRED IMPORTED_TARGET libnl-3.0 libnl-genl-3.0)
add_library(wave_wifi_native STATIC ../runner/wifi_backend.cc ../runner/wifi_radio.cc)
target_link_libraries(wave_wifi_native PUBLIC PkgConfig::GIO PkgConfig::NL)
target_include_directories(wave_wifi_native PUBLIC ../runner)
target_compile_options(wave_wifi_native PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(wifi_radio_test wifi_radio_test.cc)
target_link_libraries(wifi_radio_test PRIVATE wave_wifi_native)
add_test(NAME wifi_radio_boundary COMMAND wifi_radio_test)
add_executable(wifi_backend_test wifi_backend_test.cc)
target_link_libraries(wifi_backend_test PRIVATE wave_wifi_native)
add_test(NAME wifi_networkmanager_protocol COMMAND wifi_backend_test)
set_tests_properties(wifi_networkmanager_protocol PROPERTIES TIMEOUT 15)
