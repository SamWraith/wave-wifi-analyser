cmake_minimum_required(VERSION 3.14)
project(wave_wifi_native_tests LANGUAGES CXX)
enable_testing()
add_executable(wifi_boundary_test wifi_boundary_test.cpp)
target_compile_features(wifi_boundary_test PRIVATE cxx_std_17)
target_include_directories(wifi_boundary_test PRIVATE ../runner)
if(MSVC)
  target_compile_options(wifi_boundary_test PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(wifi_boundary_test PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME wifi_boundary_test COMMAND wifi_boundary_test)
